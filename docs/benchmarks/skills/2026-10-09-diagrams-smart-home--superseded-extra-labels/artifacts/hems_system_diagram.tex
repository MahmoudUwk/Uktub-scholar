    fill=cBox,
    line width=0.6pt

  % Energy flow (solid #D55E00)

  info/.style={
    -{Stealth[length=1.8mm, width=1.3mm]},

    line width=1.0pt,

  % Bus conductor line
  bus/.style={
    cEnergy,
    line width=1.8pt
  },
  % Group boundary
  group/.style={
    draw=black!40,
    dashed,
    rounded corners=3pt,
    inner sep=4pt
  }
}
